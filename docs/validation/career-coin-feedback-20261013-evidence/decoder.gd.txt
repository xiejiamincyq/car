extends SceneTree
## Development-side decoding only. Never included in the player package.
func _init() -> void:
	var args := OS.get_cmdline_user_args()
	if args.size() != 2:
		push_error("Expected actual PCK and its independent manifest audit")
		quit(2)
		return
	var failures: Array[String] = []
	var source_root := ProjectSettings.globalize_path("res://").replace("\\", "/").trim_suffix("/")
	if not source_root.ends_with("/tmp/career-feedback-decoder") or FileAccess.file_exists("res://scripts/main.gd"):
		failures.append("Decoder is not isolated from runtime source filesystem")
	var audit = JSON.parse_string(FileAccess.get_file_as_string(args[1]))
	if not audit is Dictionary or not audit.get("passed", false) or audit.get("logical_count", 0) != 141:
		failures.append("Actual independent 141-resource pack audit required")
	if not ProjectSettings.load_resource_pack(args[0], true):
		failures.append("Cannot mount actual PCK")
	var loaded := 0
	var textures := 0
	var music := 0
	var scenes := 0
	var main_nodes := 0
	var seen: Dictionary = {}
	if failures.is_empty():
		for item in audit.resolved:
			var path := String(item.logical)
			if seen.has(path):
				failures.append("Duplicate logical resource " + path)
				continue
			seen[path] = true
			var resource := ResourceLoader.load(path)
			if resource == null:
				failures.append("Cannot decode " + path)
				continue
			loaded += 1
			if resource is Texture2D:
				textures += 1
				if resource.get_width() <= 0 or resource.get_height() <= 0:
					failures.append("Empty texture " + path)
			elif resource is AudioStream:
				music += 1
				if resource.get_length() <= 0:
					failures.append("Empty music " + path)
			elif resource is PackedScene:
				scenes += 1
				var instance: Node = resource.instantiate()
				if instance == null:
					failures.append("Cannot instantiate " + path)
				else:
					if path == "res://scenes/main.tscn":
						main_nodes = instance.find_children("*", "", true, false).size() + 1
					instance.free()
	if loaded != 141 or textures != 68 or music != 4 or scenes != 3 or main_nodes == 0:
		failures.append("Expected exact 141 logical, 68 texture, 4 music and 3 scene coverage")
	print("UI_PACK_DECODE_COMPLETE " + JSON.stringify({"failures": failures, "loaded": loaded, "textures": textures, "music": music, "scenes": scenes, "main_nodes": main_nodes, "scope": "Actual PCK decoded in isolated matching editor engine; not normal release EXE gameplay, graphics or audible playback."}))
	quit(0 if failures.is_empty() else 1)
