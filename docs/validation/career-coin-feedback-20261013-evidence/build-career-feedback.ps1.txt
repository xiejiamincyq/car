function Invoke-UiChild($Executable,$Arguments,$Label,$Seconds=120) {
    if ($ResumeBuildRoot -and (Test-Path "$uiBuild/$Label.native.json")) {
        $uiPrevious=Get-Content "$uiBuild/$Label.native.json" -Raw|ConvertFrom-Json -AsHashtable
        if ($uiPrevious.native_exit -ne 0 -or $uiPrevious.timed_out -or (Get-FileHash "$uiBuild/$Label.stdout.log").Hash -ne $uiPrevious.stdout_sha256 -or (Get-FileHash "$uiBuild/$Label.stderr.log").Hash -ne $uiPrevious.stderr_sha256 -or ((Get-Content "$uiBuild/$Label.stdout.log" -Raw)+(Get-Content "$uiBuild/$Label.stderr.log" -Raw)) -match '(?m)SCRIPT ERROR:|Parse Error:|Assertion failed:|^ERROR:') { throw 'Cannot reuse a failed or modified native stage' }
        return $uiPrevious
    }
    $uiInfo=[Diagnostics.ProcessStartInfo]::new()
    $uiInfo.FileName=$Executable
    $uiInfo.WorkingDirectory=$uiProject
    $uiInfo.UseShellExecute=$false
    $uiInfo.CreateNoWindow=$true
    $uiInfo.RedirectStandardOutput=$true
    $uiInfo.RedirectStandardError=$true
    $uiInfo.Arguments=$Arguments
    if ($Label -in @('decode','startup','gui','coins')) {
        $uiInfo.Environment['APPDATA']=Join-Path $uiBuild ('isolated-'+$Label)
        New-Item -ItemType Directory -Path $uiInfo.Environment['APPDATA'] | Out-Null
    }
    if ($Label -like 'upgrade-*') {
        $uiInfo.Environment['APPDATA']=Join-Path $uiBuild 'isolated-upgrade'
        New-Item -ItemType Directory -Path $uiInfo.Environment['APPDATA'] -Force | Out-Null
        if ($Label -eq 'upgrade-startup') { $uiInfo.Environment['NEON_COAST_PERF_CAPTURE']='1' }
    }
    $uiChild=[Diagnostics.Process]::new()
    $uiChild.StartInfo=$uiInfo
    $uiStart=[DateTime]::UtcNow
    if (-not $uiChild.Start()) { throw 'Cannot launch owned build child' }
    $uiOutTask=$uiChild.StandardOutput.ReadToEndAsync()
    $uiErrTask=$uiChild.StandardError.ReadToEndAsync()
    $uiTimeout=$false
    while (-not $uiChild.WaitForExit(300)) { if (([DateTime]::UtcNow-$uiStart).TotalSeconds -gt $Seconds) { $uiTimeout=$true; $uiChild.Kill();$uiChild.WaitForExit();break } }
    $uiOut=$uiOutTask.GetAwaiter().GetResult();$uiErr=$uiErrTask.GetAwaiter().GetResult()
    $uiOut | Set-Content "$uiBuild/$Label.stdout.log" -Encoding utf8
    $uiErr | Set-Content "$uiBuild/$Label.stderr.log" -Encoding utf8
    $uiRecord=@{label=$Label;pid=$uiChild.Id;native_exit=$uiChild.ExitCode;timed_out=$uiTimeout;seconds=([DateTime]::UtcNow-$uiStart).TotalSeconds;stdout_sha256=(Get-FileHash "$uiBuild/$Label.stdout.log").Hash;stderr_sha256=(Get-FileHash "$uiBuild/$Label.stderr.log").Hash}
    $uiRecord | ConvertTo-Json | Set-Content "$uiBuild/$Label.native.json" -Encoding utf8
    $uiChild.Dispose()
    if ($uiTimeout -or $uiRecord.native_exit -ne 0 -or ($uiOut+$uiErr) -match '(?m)SCRIPT ERROR:|Parse Error:|Assertion failed:|^ERROR:') { throw "Native $Label failed; preserved logs in $uiBuild" }
    return $uiRecord
}
